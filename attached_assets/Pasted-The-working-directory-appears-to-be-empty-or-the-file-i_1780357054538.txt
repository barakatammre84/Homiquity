export interface URLASectionScore {
  section: string;
  /** MISMO 3.4 / URLA section reference (e.g. "1a", "2c") */
  sectionNumber: string;
  completeness: number;
  requiredFields: number;
  completedFields: number;
  missingFields: string[];
  warnings: string[];
}

export interface MISMOValidationResult {
  applicationId: string;
  overallScore: number;
  gseReady: boolean;
  ulddCompliant: boolean;
  /** HMDA data collected as required by Regulation C */
  hmdaCompliant: boolean;
  sections: URLASectionScore[];
  criticalErrors: string[];
  warnings: string[];

type FieldDef = { name: string; value: unknown; required: boolean };

/**
 * Scores a URLA section.
 *
 * Scoring: required fields carry 85 % of the weight; optional carry 15 %.
 * This keeps the score from inflating when many optional fields are present,
 * which would obscure GSE-blocking gaps.
 */
function scoreSection(
  sectionName: string,
  sectionNumber: string,
  fields: FieldDef[]
): URLASectionScore {
  const isPresent = (v: unknown) => v !== null && v !== undefined && v !== "";

  const required = fields.filter((f) => f.required);
  const optional = fields.filter((f) => !f.required);

  const completedRequired = required.filter((f) => isPresent(f.value));
  const completedOptional = optional.filter((f) => isPresent(f.value));

  const missingFields = required
    .filter((f) => !isPresent(f.value))
    .map((f) => f.name);

  const warnings = optional
    .filter((f) => !isPresent(f.value))
    .slice(0, 3)
    .map((f) => `${f.name} is recommended but not provided`);

  const reqScore =
    required.length > 0
      ? (completedRequired.length / required.length) * 0.85
      : 0.85;
  const optScore =
    optional.length > 0
      ? (completedOptional.length / optional.length) * 0.15
      : 0.15;

  return {
    section: sectionName,
    sectionNumber,
    completeness: Math.round((reqScore + optScore) * 100),
    requiredFields: required.length,
    completedFields: completedRequired.length,
    missingFields,
    warnings,
  };
}

/** URLA Section 1a — Borrower Information */
function scorePersonalInfo(
  personalInfo: UrlaPersonalInfo | null | undefined
): URLASectionScore {
  return scoreSection("Personal Information", "1a", [
    { name: "FirstName", value: personalInfo?.firstName, required: true },
    { name: "LastName", value: personalInfo?.lastName, required: true },
    // TaxpayerIdentifierValue (SSN) — GSE hard requirement

    { name: "MaritalStatusType", value: personalInfo?.maritalStatus, required: true },
    { name: "ContactPointEmailValue", value: personalInfo?.email, required: false },

/** URLA Sections 1b-1e — Employment & Income */
function scoreEmployment(employment: EmploymentHistory[]): URLASectionScore {
  const current = employment.find((e) => e.employmentType === "current");

  return scoreSection("Employment & Income", "1b-1e", [
    { name: "EmployerName", value: current?.employerName, required: true },
    { name: "PositionDescription", value: current?.positionTitle, required: true },
    { name: "EmploymentStartDate", value: current?.startDate, required: true },
    // BasePayAmount maps to MISMO BaseIncome
    { name: "BasePayAmount", value: current?.monthlyIncomeOrLoss, required: true },
    { name: "EmployerPhone", value: current?.employerPhone, required: false },
    { name: "EmployerAddressLineText", value: current?.employerStreet, required: false },
    // SelfEmployedIndicator — drives Schedule C / K-1 doc requirements
    { name: "SelfEmployedIndicator", value: current?.isSelfEmployed, required: true },
    // GSE requires 2-year history; flag when prior employer absent
    {
      name: "PriorEmployerName (2yr history)",
      value: employment.find((e) => e.employmentType === "previous")?.employerName,
      required: false,

  ]);
}

/** URLA Section 1f — Other Income */
function scoreOtherIncome(application: LoanApplication): URLASectionScore {
  // If the application carries any declared other-income source, validate it.
  const hasOtherIncome = (application as any).otherIncomeAmount;

  return scoreSection("Other Income", "1f", [
    {
      name: "OtherIncomeSourceTypeDescription",
      value: (application as any).otherIncomeSourceType,
      required: !!hasOtherIncome,
    },
    {
      name: "OtherIncomeMonthlyAmount",
      value: hasOtherIncome,
      required: false,
    },
  ]);
}

/** URLA Section 2a — Assets */
function scoreAssets(assets: UrlaAsset[]): URLASectionScore {
  const hasAssets = assets.length > 0;
  const totalValue = assets.reduce(
    (sum, a) => sum + Number(a.cashOrMarketValue || 0),
    0
  );
  const first = assets[0];

  return scoreSection("Assets & Accounts", "2a", [
    { name: "AtLeastOneAssetDeclared", value: hasAssets ? "yes" : null, required: true },
    { name: "TotalAssetAmount > $0", value: totalValue > 0 ? "yes" : null, required: true },
    { name: "DepositoryAccountType", value: first?.accountType, required: hasAssets },
    { name: "FinancialInstitutionName", value: first?.financialInstitution, required: hasAssets },
    { name: "AssetCurrentActualValue", value: first?.cashOrMarketValue, required: hasAssets },
    // Gift funds must be sourced per GSE guidelines
    {
      name: "GiftFundsSourceType",
      value: assets.find((a) => a.accountType === "Gift")?.financialInstitution,
      required: false,
    },
  ]);
}

/**
 * Adds N business days (Mon-Fri, no holiday awareness) to a date.
 * TRID defines "business day" as all calendar days except Sundays and federal
 * public holidays — this approximation excludes only weekends.
 */
function addBusinessDays(date: Date, days: number): Date {
  const result = new Date(date);
  let added = 0;
  while (added < days) {
    result.setDate(result.getDate() + 1);
    const dow = result.getDay();
    if (dow !== 0 && dow !== 6) added++;
  }
  return result;
}

export async function validateMISMOCompleteness(
  applicationId: string
): Promise<MISMOValidationResult> {
  const application = await storage.getLoanApplication(applicationId);
  if (!application) throw new Error("Application not found");

  const urlaData = await storage.getCompleteUrlaData(applicationId);
  const conditions = await storage.getLoanConditionsByApplication(applicationId);
  const documents = await storage.getDocumentsByApplication(applicationId);

  // Score all MISMO 3.4 / URLA 2021 sections
  const sections: URLASectionScore[] = [
    scorePersonalInfo(urlaData.personalInfo),
    scoreEmployment(urlaData.employmentHistory),
    scoreOtherIncome(application),
    scoreAssets(urlaData.assets),
    scoreLiabilities(urlaData.liabilities),
    scoreReo(application),
    scorePropertyInfo(application, urlaData.propertyInfo),
    scoreLoanInfo(application),
    scoreLoanOriginator(application),
    scoreDeclarations(urlaData.declarations),
    scoreMilitaryService(application),
    scoreHmda(application),
  ];

  const overallScore = Math.round(
    sections.reduce((sum, s) => sum + s.completeness, 0) / sections.length
  );

  const criticalErrors: string[] = [];
  const warnings: string[] = [];

  for (const section of sections) {
    for (const field of section.missingFields) {
      criticalErrors.push(`${section.section} (${section.sectionNumber}): ${field} is required`);
    }
    for (const w of section.warnings) {
      warnings.push(`${section.section}: ${w}`);
    }
  }

  // Missing documents from outstanding conditions
  const outstandingConditions = conditions.filter((c) => c.status === "outstanding");
  const requiredDocTypes = [
    ...new Set(outstandingConditions.flatMap((c) => c.requiredDocumentTypes ?? [])),
  ];
  const uploadedDocTypes = new Set(documents.map((d) => d.documentType));
  const missingDocuments = requiredDocTypes.filter((t) => !uploadedDocTypes.has(t));

  // TRID — LE due 3 business days from application receipt; CD due 3 business
  // days before consummation (tracked separately when closing date is known).
  const receivedAt = application.createdAt ? new Date(application.createdAt) : new Date();
  const leDueDate = addBusinessDays(receivedAt, 3);

  const app = application as any;
  const closingDate: Date | null = app.closingDate ? new Date(app.closingDate) : null;
  const cdDueDate: Date | null = closingDate
    ? new Date(closingDate.getTime() - 3 * 24 * 60 * 60 * 1000)
    : null;

  // GSE readiness: 90 % overall, zero critical errors, no missing docs,
  // and the key rate-lock / AUS / originator fields present.
  const hasNmls = !!(app.loanOriginator?.nmls);
  const hasAmortType = !!(app.amortizationType);
  const gseReady =
    overallScore >= 90 &&
    criticalErrors.length === 0 &&
    missingDocuments.length === 0 &&
    hasNmls &&
    hasAmortType;

  // ULDD compliance: 80 % overall + key personal/loan sections strong
  const personalSection = sections.find((s) => s.sectionNumber === "1a");
  const loanSection = sections.find((s) => s.sectionNumber === "4");
  const ulddCompliant =
    overallScore >= 80 &&
    (personalSection?.completeness ?? 0) >= 90 &&
    (loanSection?.completeness ?? 0) >= 90;

  // HMDA compliance: demographic section fully answered (or refusals recorded)
  const hmdaSection = sections.find((s) => s.sectionNumber === "7");
  const hmdaCompliant = (hmdaSection?.completeness ?? 0) >= 90;

  // ULDD Phase 3 checklist — individual flags surfaced for loan pipeline UI
  const ulddChecklist = {
    ausIdentifier: !!(app.ausCasefileId),
    miCoveragePresent: !!(app.miCoveragePercent),
    escrowIndicator: app.escrowIndicator !== undefined && app.escrowIndicator !== null,
    propertyValuationType: !!(app.propertyValuationType),
    subjectPropertyCltv: !!(app.cltvRatio),
  };

  return {
    applicationId,
    overallScore,
    gseReady,
    ulddCompliant,
    hmdaCompliant,
    sections,
    criticalErrors,
    warnings,
    missingDocuments,
    tridStatus: {
      leRequired: application.status !== "draft",
      leDueDate,
      leIssued: !!(app.leIssuedDate),
      cdRequired: application.status === "clear_to_close",
      cdDueDate,
      cdIssued: !!(app.cdIssuedDate),
      revisedLeRequired: !!(app.revisedLeRequired),
    },
    ulddChecklist,
  };
}

export async function getApplicationValidationSummary(applicationId: string) {
  const validation = await validateMISMOCompleteness(applicationId);

  return {
    applicationId,
    score: validation.overallScore,
    gseReady: validation.gseReady,
    ulddCompliant: validation.ulddCompliant,
    hmdaCompliant: validation.hmdaCompliant,
    criticalCount: validation.criticalErrors.length,
    warningCount: validation.warnings.length,
    missingDocsCount: validation.missingDocuments.length,
    ulddChecklist: validation.ulddChecklist,
    sections: validation.sections.map((s) => ({
      name: s.section,
      number: s.sectionNumber,
      score: s.completeness,
      complete: s.completeness >= 90,
    })),
  };
}

export async function getBatchValidationStatus(applicationIds: string[]) {
  const results = await Promise.all(
    applicationIds.map((id) =>
      getApplicationValidationSummary(id).catch(() => null)
    )
  );
  return results.filter((r) => r !== null);
}

That's a comprehensive audit. Here are the findings organized by what to act on now vs. later:

Remaining Gaps — Prioritized
Blocking for GSE delivery (fix before going live)
1. Co-applicant entirely absent
Every section only scores the primary borrower. ECOA (Reg B §202.5), HMDA §1003.4(a)(23–26), and URLA §1a–5 all require co-applicant personal info, employment, assets, liabilities, declarations, and demographics to be collected and scored independently. This is the single largest gap.

2. Two scoring logic bugs that inflate scores

scoreSection() awards optionalWeight = 0.15 even when zero optional fields are present (optional.length === 0 hits the else branch). A section with only required fields and all of them missing still scores 15.
scoreLiabilities() hardcodes LiabilitiesDisclosedIndicator: "yes" unconditionally — this field should come from actual borrower data, not be assumed.
Same issue in scoreReo() — REODisclosedIndicator is hardcoded "yes" regardless of whether the borrower actually disclosed anything.
3. addBusinessDays() is wrong for TRID
It excludes weekends but not federal holidays. TRID 12 CFR 1026.19 defines business days as all calendar days except Sundays and federal public holidays. A LE due on Christmas or July 4 will be calculated incorrectly, which is a regulatory violation. Need to embed a federal holiday list.

4. CD due date calculation is wrong
closingDate.getTime() - 3 * 24 * 60 * 60 * 1000 subtracts 3 calendar days, not 3 business days. The CD waiting period under TRID is 3 business days before consummation — same addBusinessDays() logic (in reverse) applies.

5. ARM fields completely absent from loan scoring
If amortizationType is ARM, TRID 12 CFR 1026.19(b)(2), HMDA §1003.4(a)(31), and GSE delivery all require: index type, margin, initial rate, periodic cap, lifetime cap, adjustment frequency, rate floor. None are scored or validated.

6. HMDA LAR action-taken code and denial reasons missing
Reg C §1003.4(a)(16–17) requires ActionTakenType (originated/approved/denied/withdrawn/incomplete) and if denied, 2+ denial reason codes. These don't exist anywhere in the validation.

7. ATR/QM — points & fees cap not validated
12 CFR 1026.43(e)(1)(i): points and fees must be ≤ 3% of loan amount for QM status. The code captures loan amount but never calculates or validates this. A loan can pass your GSE check and still be non-QM.

8. No HMDA rate spread / APOR
§1003.4(a)(29): rate spread = note rate − APOR (Average Prime Offer Rate). Required for every originated loan. Neither the APOR reference rate nor the spread is captured.

Important but not immediately blocking
9. Overall score uses unweighted average across all 12 sections
A missing SSN (section 1a) dilutes to ~8% of the overall score. Sections 1a, 4, and 5 (Declarations) should be gating — if any required field in those sections is missing, gseReady should be false regardless of overall score.

10. warnings array is capped at 3 per section
.slice(0, 3) silently drops additional warnings. The caller has no way to know there are more. Either remove the cap or add a warningCount to the section score.

11. scoreEmployment() marks prior employer as optional when < 2 years at current
GSE guides require a full 2-year history. The code should detect current.startDate within 2 years and flip PriorEmployerName to required: true.

12. No QM type flag
The result should surface whether the loan is QM Safe Harbor, QM Rebuttably Presumed, or Non-QM. Downstream systems need this for secondary market eligibility.

13. Community property states
If propertyState or currentState is CA, TX, AZ, NV, WA, ID, NM, LA, or WI, spouse income/debt rules differ and HMDA co-applicant handling changes. No detection exists.

14. ULDD checklist — miCoveragePresent doesn't enforce MI when LTV > 80%
It only checks if the percentage field is populated. For conventional loans with LTV > 80%, MI is mandatory — the checklist should fail if LTV > 80% and miCoveragePercent is absent.

Minor / structural
Heavy as any casting throughout should be replaced with a proper extended interface (e.g., LoanApplicationExtended) to get compile-time safety on the new fields
scoreSection() warning cap of 3 should be removed or exposed as additionalWarnings: number
getBatchValidationStatus() swallows errors silently — consider returning { id, error } for failed ones so callers know which apps failed vs. which had no errors
